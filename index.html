namespace App\Leads\Application;

final class LeadProcessor
{
    public function __construct(
        private readonly LeadRepository   $leads,
        private readonly EventPublisher   $bus,
        private readonly ScoringService   $scoring,
    ) {}

    public function __invoke(IngestLead $cmd): void
    {
        $lead  = Lead::fromCommand($cmd);
        $score = $this->scoring->evaluate($lead);

        $this->leads->save($lead->withScore($score));
        $this->bus->publish(new LeadIngested($lead->id()));
    }
}
namespace App\Http\Controllers;

use App\Models\Lead;
use App\Services\ScoringService;
use App\Events\LeadIngested;
use Illuminate\Http\Request;

class LeadController extends Controller
{
    public function __construct(
        private readonly ScoringService $scoring,
    ) {}

    public function ingest(Request $request)
    {
        $lead  = Lead::create($request->validated());
        $score = $this->scoring->evaluate($lead);

        $lead->update(['score' => $score]);
        event(new LeadIngested($lead->id));

        return response()->json(['id' => $lead->id, 'score' => $score]);
    }
}

package leads

func HandleIngest(scoring ScoringService, repo Repository, bus EventPublisher) http.HandlerFunc {
    return func(w http.ResponseWriter, r *http.Request) {
        var cmd IngestLead
        if err := json.NewDecoder(r.Body).Decode(&cmd); err != nil {
            http.Error(w, err.Error(), http.StatusBadRequest)
            return
        }

        lead := NewLeadFromCommand(cmd)
        score, err := scoring.Evaluate(r.Context(), lead)
        if err != nil {
            http.Error(w, err.Error(), http.StatusInternalServerError)
            return
        }

        lead = lead.WithScore(score)
        repo.Save(r.Context(), lead)
        bus.Publish(r.Context(), LeadIngested{ID: lead.ID})

        json.NewEncoder(w).Encode(lead)
    }
}

    r.POST("/leads", func(c *gin.Context) {
        var cmd IngestLead

            c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
            return
        }

        lead := NewLeadFromCommand(cmd)
        score, err := scoring.Evaluate(c, lead)
        if err != nil {
            c.JSON(http.StatusInternalServerError, gin.H{"error": err.Error()})
            return
        }

        lead = lead.WithScore(score)
        repo.Save(c, lead)
        bus.Publish(c, LeadIngested{ID: lead.ID})

        c.JSON(http.StatusOK, lead)
    })
}

  COUNT(DISTINCT l.id)                  AS customers,
  SUM(s.amount)                         AS mrr,
  ROUND(AVG(s.amount), 2)               AS arpu
FROM leads l
JOIN subscriptions s ON s.lead_id = l.id
WHERE l.status = 'converted' AND s.active = 1
GROUP BY cohort
HAVING COUNT(DISTINCT l.id) > 50
ORDER BY cohort DESC
LIMIT 24;
WITH cohorts AS (
  SELECT id,
         date_trunc('month', created_at) AS cohort
  FROM   leads
  WHERE  status = 'converted'
)
SELECT c.cohort,
       COUNT(DISTINCT c.id)                      AS customers,
       SUM(s.amount)                                 AS mrr,
       ROUND(AVG(s.amount)::numeric, 2)          AS arpu,
       RANK() OVER (ORDER BY SUM(s.amount) DESC) AS rnk
FROM      cohorts       c
JOIN      subscriptions s ON s.lead_id = c.id
WHERE     s.active
GROUP BY  c.cohort
HAVING    COUNT(DISTINCT c.id) > 50
ORDER BY  c.cohort DESC
LIMIT     24;

    SUM(s.amount)                  AS mrr,
    ROUND(AVG(s.amount), 2)        AS arpu
  FROM leads l

  WHERE l.status = 'converted' AND s.active = 1
